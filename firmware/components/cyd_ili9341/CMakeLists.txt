idf_component_register(SRCS "cyd_ili9341.c" INCLUDE_DIRS "include" REQUIRES esp_lcd esp_driver_spi esp_driver_gpio freertos heap)
if(NOT DEFINED HW3_LCD_BGR)
  set(HW3_LCD_BGR 0)
endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE LCD_BGR=${HW3_LCD_BGR})
target_compile_options(${COMPONENT_LIB} PRIVATE -Wall -Wextra -Werror -std=gnu17)

# HW4-SPI40: component discovery without root cache uses 20 MHz.
if(NOT DEFINED CYD_LCD_SPI_HZ)
  set(CYD_LCD_SPI_HZ "20000000")
endif()
if(NOT (CYD_LCD_SPI_HZ STREQUAL "20000000" OR CYD_LCD_SPI_HZ STREQUAL "40000000"))
  message(FATAL_ERROR "HW4-SPI40 20/40 MHz only")
endif()
if(NOT LOCO_CYD_PROFILE STREQUAL "CONSOLE" AND CYD_LCD_SPI_HZ STREQUAL "40000000")
  message(FATAL_ERROR "HW4-SPI40 40 MHz CONSOLE only")
endif()
target_compile_definitions(${COMPONENT_LIB} PRIVATE LCD_CLOCK_HZ=${CYD_LCD_SPI_HZ}U)
